#include <queue>
#include <mutex>
#include <condition_variable>

struct DataPacket {
    std::vector<UCHAR> data;
};

class DataQueue {
private:
    std::queue<DataPacket> queue;
    std::mutex mutex;
    std::condition_variable cv;
    bool active = true;

public:
    void push(std::vector<UCHAR>&& packet) {
        std::lock_guard<std::mutex> lock(mutex);
        queue.push(std::move(packet));
        cv.notify_one(); // Wake up the processing thread
    }

    bool pop(DataPacket& packet) {
        std::unique_lock<std::mutex> lock(mutex);
        cv.wait(lock, [this] { return !queue.empty() || !active; });
        if (queue.empty() && !active) return false;
        packet = std::move(queue.front());
        queue.pop();
        return true;
    }

    void shutdown() {
        std::lock_guard<std::mutex> lock(mutex);
        active = false;
        cv.notify_all();
    }
};

// The Buffer Pool Implementation
#include <vector>
#include <mutex>
#include <stack>

class BufferPool {
private:
    std::stack<UCHAR*> freeBuffers;
    std::mutex mutex;
    size_t bufferSize;

public:
    BufferPool(size_t count, size_t size) : bufferSize(size) {
        for (size_t i = 0; i < count; ++i) {
            // Use _aligned_malloc for better performance with USB 3.0 DMA
            freeBuffers.push((UCHAR*)_aligned_malloc(size, 4096));
        }
    }

    UCHAR* acquire() {
        std::lock_guard<std::mutex> lock(mutex);
        if (freeBuffers.empty()) return nullptr; // Pool exhausted
        UCHAR* buf = freeBuffers.top();
        freeBuffers.pop();
        return buf;
    }

    void release(UCHAR* buf) {
        std::lock_guard<std::mutex> lock(mutex);
        freeBuffers.push(buf);
    }

    ~BufferPool() {
        while (!freeBuffers.empty()) {
            _aligned_free(freeBuffers.top());
            freeBuffers.pop();
        }
    }
};

3. Using FT_ReadPipeEx and FT_GetOverlappedResults
When using FT_ReadPipeEx, it returns FT_IO_PENDING immediately. You then use FT_GetOverlappedResult 
(or FT_W32_GetOverlappedResult) in the completion handler. 

// Example: Asynchronous Read using FT_ReadPipeEx
auto context = std::make_shared<D3XXOverlappedContext>();
context->overlapped.hEvent = CreateEvent(NULL, TRUE, FALSE, NULL);
context->pipeId = 0x82; // Example Pipe ID
context->buffer.resize(1024);

// Initiate async read
ftStatus = FT_ReadPipeEx(ftHandle, 
                         context->pipeId, 
                         context->buffer.data(), 
                         context->buffer.size(), 
                         &context->overlapped);

if (ftStatus != FT_IO_PENDING && ftStatus != FT_OK) {
    // Handle error
    CloseHandle(context->overlapped.hEvent);
}

// Associate the event with asio to handle completion
auto waiter = std::make_shared<boost::asio::windows::object_handle>(io_ctx, context->overlapped.hEvent);
waiter->async_wait([context, waiter, ftHandle](const boost::system::error_code& ec) {
    if (!ec) {
        DWORD bytesTransferred = 0;
        // Retrieve result
        BOOL success = FT_W32_GetOverlappedResult(ftHandle, 
                                                 &context->overlapped, 
                                                 &bytesTransferred, 
                                                 FALSE);
        if (success) {
            std::cout << "Read " << bytesTransferred << " bytes from pipe " << context->pipeId << std::endl;
        }
    }
    // Cleanup
    CloseHandle(context->overlapped.hEvent);
});

io_ctx.run(); // Run the Asio event loop

To pass additional parameters, the best approach is to encapsulate the OVERLAPPED structure within a larger custom struct

Completion Handler: Use lambda captures ([context, ... ]) to bring the context struct into the async_wait handler.
Reusing Structure: Crucial: Do not reuse the D3XXOverlappedContext or the OVERLAPPED structure until the previous async operation is
guaranteed to have finished (i.e., after the async_wait handler completes).

#include <boost/asio.hpp>
#include <windows.h>
#include <ftd3xx.h>
#include <iostream>

// 1. Custom structure to pass additional parameters
struct MyOverlapped : OVERLAPPED {
    void* custom_data;
    // Callback function pointer
    void (*on_complete)(MyOverlapped*, DWORD, DWORD); 

    MyOverlapped() {
        memset(this, 0, sizeof(OVERLAPPED));
        hEvent = CreateEvent(NULL, TRUE, FALSE, NULL); // Manual reset
    }
    ~MyOverlapped() { CloseHandle(hEvent); }
};

// 2. Callback function
void OnReadComplete(MyOverlapped* ov, DWORD errorCode, DWORD bytesTransferred) {
    std::cout << "Read completed. Bytes: " << bytesTransferred << " Error: " << errorCode << std::endl;
}

// ... Inside main ...
FT_HANDLE ftHandle;
FT_Create(0, FT_OPEN_BY_INDEX, &ftHandle);

// Wrap Handle in Boost.Asio
boost::asio::io_context io_ctx;
boost::asio::windows::stream_handle ftdi_stream(io_ctx, (HANDLE)ftHandle);

// 3. Initiate Read with Extended Info
MyOverlapped* ov = new MyOverlapped();
ov->custom_data = nullptr;
ov->on_complete = OnReadComplete;

unsigned char buffer[1024];
DWORD bytesRead = 0;
FT_ReadPipeEx(ftHandle, 0, buffer, sizeof(buffer), &bytesRead, ov);

// 4. Handle results later (e.g., in a separate thread or via ASIO)
// ... In a waiting thread ...
if (FT_GetOverlappedResult(ftHandle, ov, &bytesRead, TRUE)) {
    ov->on_complete(ov, 0, bytesRead);
}

Event Handling: FT_GetOverlappedResult relies on hEvent inside the OVERLAPPED structure being set properly.
Cleanup: Use FT_ReleaseOverlapped to properly clean up the OVERLAPPED structure after the operation is complete.
Asio Integration: While you can use FT_ReadPipeEx directly, you can also assign the stream_handle to io_context and use async_read_some for more advanced Asio integration.
